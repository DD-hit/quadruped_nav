cmake_minimum_required(VERSION 3.16)
project(quadruped_web)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(octomap REQUIRED)
add_executable(octomap_web_export src/octomap_web_export.cpp)
target_include_directories(octomap_web_export PRIVATE ${OCTOMAP_INCLUDE_DIRS})
target_link_libraries(octomap_web_export ${OCTOMAP_LIBRARIES})
install(TARGETS octomap_web_export DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/web_server.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY web config licenses DESTINATION share/${PROJECT_NAME})
install(FILES NOTICE README.md DESTINATION share/${PROJECT_NAME})
ament_package()
