cmake_minimum_required(VERSION 3.16)
project(quadruped_navigation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters)
find_package(OpenCV REQUIRED)
find_package(Open3D REQUIRED)
find_package(octomap REQUIRED)
set(ros_deps rclcpp sensor_msgs geometry_msgs nav_msgs quadruped_interfaces octomap_msgs
 visualization_msgs std_msgs cv_bridge pcl_conversions message_filters tf2 tf2_geometry_msgs tf2_ros tf2_sensor_msgs)
foreach(dep ${ros_deps})
  find_package(${dep} REQUIRED)
endforeach()
add_library(scan_core STATIC
 src/bspline_optimizer.cpp src/uniform_bspline.cpp src/gradient_descent_optimizer.cpp
 src/dyn_a_star.cpp src/grid_map.cpp src/raycast.cpp src/planning_visualization.cpp
 src/polynomial_traj.cpp src/planner_manager.cpp)
target_include_directories(scan_core PUBLIC include ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(scan_core ${ros_deps})
target_link_libraries(scan_core octomap Eigen3::Eigen ${PCL_LIBRARIES} ${OpenCV_LIBS})
add_executable(local_planner src/scan_planner_node.cpp src/scan_replan_fsm.cpp)
ament_target_dependencies(local_planner ${ros_deps})
target_link_libraries(local_planner scan_core)
add_executable(trajectory_controller src/closed_loop_controller.cpp)
ament_target_dependencies(trajectory_controller ${ros_deps})
target_link_libraries(trajectory_controller scan_core)
add_executable(global_planner src/global_planner.cpp)
ament_target_dependencies(global_planner ${ros_deps})
target_link_libraries(global_planner octomap)
set(core_targets global_planner local_planner trajectory_controller)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/frame_adapter.cpp")
  add_executable(frame_adapter src/frame_adapter.cpp)
  ament_target_dependencies(frame_adapter ${ros_deps})
  target_link_libraries(frame_adapter Eigen3::Eigen)
  list(APPEND core_targets frame_adapter)
endif()
foreach(name pcd_to_octomap_node octomap_to_occupied_markers_node octomap_to_cloud_node)
 add_executable(${name} src/map_tools/${name}.cpp)
 ament_target_dependencies(${name} ${ros_deps})
 target_include_directories(${name} PRIVATE /usr/include/eigen3)
 target_link_libraries(${name} octomap Open3D::Open3D)
 list(APPEND core_targets ${name})
endforeach()
install(TARGETS ${core_targets} DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(quadruped_navigation_tools)
install(DIRECTORY scripts/ DESTINATION lib/${PROJECT_NAME} USE_SOURCE_PERMISSIONS PATTERN "__pycache__" EXCLUDE)
install(DIRECTORY launch config licenses DESTINATION share/${PROJECT_NAME})
install(FILES README.md NOTICE migration_sources.json DESTINATION share/${PROJECT_NAME})
ament_package()
